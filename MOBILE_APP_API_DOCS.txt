11. READY-TO-USE FLUTTER (DART) SERVICE CLASS

import 'dart:convert';
import 'package:http/http.dart' as http;

class SchoolApiService {
  static const String baseUrl = 'http://10.0.2.2/school/api'; // For Android Emulator
  static const String apiKey = 'sk_live_demo_flutter_app_key_2026';
  static String? bearerToken;

  static Map<String, String> get defaultHeaders => {
    'Content-Type': 'application/json',
    'X-API-KEY': apiKey,
    if (bearerToken != null) 'Authorization': 'Bearer $bearerToken',
  };

  // 1. Health Ping
  static Future<bool> ping() async {
    final res = await http.get(Uri.parse('$baseUrl/status'), headers: defaultHeaders);
    return res.statusCode == 200;
  }

  // 2. Login
  static Future<Map<String, dynamic>> login(String identifier, String password) async {
    final res = await http.post(
      Uri.parse('$baseUrl/login'),
      headers: defaultHeaders,
      body: jsonEncode({
        'username': identifier,
        'password': password,
        'device_platform': 'android',
      }),
    );
    final data = jsonDecode(res.body);
    if (data['success'] == true) {
      bearerToken = data['data']['token'];
    }
    return data;
  }

  // 3. Mark Attendance
  static Future<bool> saveAttendance(int classId, int sectionId, String date, List<Map<String, dynamic>> students) async {
    final res = await http.post(
      Uri.parse('$baseUrl/save-attendance'),
      headers: defaultHeaders,
      body: jsonEncode({
        'class_id': classId,
        'section_id': sectionId,
        'date': date,
        'students': students,
      }),
    );
    final data = jsonDecode(res.body);
    return data['success'] == true;
  }
}
